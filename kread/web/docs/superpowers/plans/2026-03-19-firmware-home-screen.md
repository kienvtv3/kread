# Firmware Home Screen Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the Home screen on ESP32-C3 firmware with book display (cover, title, author, progress), Left/Right book navigation, battery indicator, and help bar — matching the web emulator.

**Architecture:** UI assets are embedded in firmware flash as C arrays (generated from .kp files). Books are .kb files on SD card, parsed by kb_reader.c. Home screen blits pre-rendered assets to dual framebuffers (BW + GS) for 4-level grayscale display.

**Tech Stack:** C (ESP-IDF), PlatformIO, SSD1677 E-Ink driver, LZ4 decompression

**Spec:** `docs/superpowers/specs/2026-03-19-firmware-home-screen-design.md`

**Web dir:** `C:\Projects\kread-master\kread\web`

## Task 1: Asset Header Generator Script

**Goal:** Create `gen-asset-headers.mjs` that converts all .kp files into C source files for firmware embedding.

**Files:**
- Create: `web/scripts/gen-asset-headers.mjs`
- Modify: `web/package.json`

- [ ] **Step 1: Create the generator script**

Create `web/scripts/gen-asset-headers.mjs`:

```js
#!/usr/bin/env node
/**
 * Convert .kp asset files → C source for firmware embedding.
 *
 * Reads all .kp files from resources/assets/{en,vi}/
 * Generates:
 *   firmware/include/ui_assets.h  — enum + struct + externs
 *   firmware/src/ui_assets_data.c — const byte arrays + tables
 *
 * Usage: node scripts/gen-asset-headers.mjs
 */

import { readFileSync, writeFileSync, readdirSync } from 'fs'
import { resolve, dirname, basename } from 'path'
import { fileURLToPath } from 'url'

const __dirname = dirname(fileURLToPath(import.meta.url))
const ASSETS_DIR = resolve(__dirname, '../resources/assets')
const FW_INCLUDE = resolve(__dirname, '../../firmware/include')
const FW_SRC = resolve(__dirname, '../../firmware/src')

const LANGUAGES = ['en', 'vi']

// Read all asset keys from the first language dir
const assetKeys = readdirSync(resolve(ASSETS_DIR, 'en'))
  .filter(f => f.endsWith('.kp'))
  .map(f => f.replace('.kp', ''))
  .sort()

// Convert key like "help.read" → C identifier "HELP_READ"
function keyToEnum(key) {
  return 'ASSET_' + key.toUpperCase().replace(/\./g, '_')
}

// Convert key like "help.read" → C variable name "_en_help_read"
function keyToVar(lang, key) {
  return `_${lang}_${key.replace(/\./g, '_')}`
}

// ── Generate header ──

let header = `// ui_assets.h — AUTO-GENERATED by gen-asset-headers.mjs. Do not edit.
#pragma once
#include <stdint.h>

typedef struct {
    const uint8_t *data;  // KP v2 blob in flash (DROM)
    uint16_t w, h;        // pixel dimensions
    uint32_t size;         // blob size in bytes
} ui_asset_t;

// Asset index — shared across all languages
typedef enum {
`

for (let i = 0; i < assetKeys.length; i++) {
  header += `    ${keyToEnum(assetKeys[i])} = ${i},\n`
}
header += `    ASSET_COUNT = ${assetKeys.length}\n`
header += `} ui_asset_id_t;\n\n`

for (const lang of LANGUAGES) {
  header += `extern const ui_asset_t UI_ASSETS_${lang.toUpperCase()}[];\n`
}
header += `\n// Runtime language pointer\nextern const ui_asset_t *ui_lang;\n`
header += `void ui_set_language(const ui_asset_t *table);\n`

writeFileSync(resolve(FW_INCLUDE, 'ui_assets.h'), header)
console.log(`Generated ui_assets.h (${assetKeys.length} assets)`)

// ── Generate data ──

let data = `// ui_assets_data.c — AUTO-GENERATED by gen-asset-headers.mjs. Do not edit.\n`
data += `#include "ui_assets.h"\n\n`
data += `const ui_asset_t *ui_lang = NULL; // set in main.c\n`
data += `void ui_set_language(const ui_asset_t *table) { ui_lang = table; }\n\n`

for (const lang of LANGUAGES) {
  data += `// ── ${lang.toUpperCase()} ──\n\n`

  // Byte arrays
  for (const key of assetKeys) {
    const filePath = resolve(ASSETS_DIR, lang, `${key}.kp`)
    const bytes = readFileSync(filePath)
    const varName = keyToVar(lang, key)
    data += `static const uint8_t ${varName}[] = {\n    `
    const hexBytes = Array.from(bytes).map(b => '0x' + b.toString(16).padStart(2, '0'))
    // Wrap lines at 16 bytes
    for (let i = 0; i < hexBytes.length; i += 16) {
      if (i > 0) data += '    '
      data += hexBytes.slice(i, i + 16).join(',') + ',\n'
    }
    data += `};\n\n`
  }

  // Table
  data += `const ui_asset_t UI_ASSETS_${lang.toUpperCase()}[] = {\n`
  for (const key of assetKeys) {
    const filePath = resolve(ASSETS_DIR, lang, `${key}.kp`)
    const bytes = readFileSync(filePath)
    // Parse KP header for w, h
    const w = bytes[4] | (bytes[5] << 8)
    const h = bytes[6] | (bytes[7] << 8)
    const varName = keyToVar(lang, key)
    data += `    [${keyToEnum(key)}] = { ${varName}, ${w}, ${h}, sizeof(${varName}) },\n`
  }
  data += `};\n\n`
}

writeFileSync(resolve(FW_SRC, 'ui_assets_data.c'), data)
const sizeKB = (data.length / 1024).toFixed(1)
console.log(`Generated ui_assets_data.c (${sizeKB} KB source)`)
```

- [ ] **Step 1: Create header**

Create `firmware/include/kp_decode.h`:
```c
#pragma once
#include <stdint.h>
#include <stdbool.h>

/**
 * Blit a KP v2 asset (from flash or RAM) to dual framebuffers.
 *
 * KP v2 format: [magic:4][w:2][h:2][depth:1][comp:1][dataSize:4][flags:1][reserved:1]
 * Data (bit_depth=2): [lsb_size:4][lsb_data][msb_size:4][msb_data]
 *
 * @param kp_data  Pointer to KP blob (flash or RAM)
 * @param kp_size  Total blob size
 * @param fb_bw    BW framebuffer (landscape, 48KB)
 * @param fb_gs    GS framebuffer (landscape, 48KB), NULL for B&W only
 * @param dst_x    Portrait X destination
 * @param dst_y    Portrait Y destination
 * @return true on success
 */
bool kp_blit(const uint8_t *kp_data, uint32_t kp_size,
             uint8_t *fb_bw, uint8_t *fb_gs,
             int dst_x, int dst_y);

/**
 * Blit a KP asset centered horizontally at given bottom Y.
 * x = (480 - asset_w) / 2
 * y = bottom_y - asset_h
 */
bool kp_blit_centered(const uint8_t *kp_data, uint32_t kp_size,
                       uint8_t *fb_bw, uint8_t *fb_gs,
                       int center_x, int bottom_y);

/**
 * Parse KP header to get dimensions without blitting.
 */
bool kp_get_dimensions(const uint8_t *kp_data, uint16_t *w, uint16_t *h);
```

- [ ] **Step 2: Create implementation**

Create `firmware/src/kp_decode.c`:
```c
#include "kp_decode.h"
#include "blit.h"
#include "lz4.h"
#include <string.h>

// KP v2 magic: 'K' 'P' 0x00 0x02
static const uint8_t KP_MAGIC[] = { 0x4B, 0x50, 0x00, 0x02 };

// Temp buffer for LZ4 decompression (largest plane: 800*480/8 = 48000 bytes)
// For UI assets (small), this is way more than needed.
// For page-size assets, we decompress directly.
#define KP_DECOMP_BUF_SIZE 48000
static uint8_t decomp_buf[KP_DECOMP_BUF_SIZE];

static inline uint16_t read_u16(const uint8_t *p) { return p[0] | (p[1] << 8); }
static inline uint32_t read_u32(const uint8_t *p) { return p[0] | (p[1] << 8) | (p[2] << 16) | (p[3] << 24); }

bool kp_blit(const uint8_t *kp_data, uint32_t kp_size,
             uint8_t *fb_bw, uint8_t *fb_gs,
             int dst_x, int dst_y) {
    if (kp_size < 16) return false;
    if (memcmp(kp_data, KP_MAGIC, 4) != 0) return false;

    uint16_t w = read_u16(kp_data + 4);
    uint16_t h = read_u16(kp_data + 6);
    uint8_t bit_depth = kp_data[8];
    uint8_t compression = kp_data[9];

    if (bit_depth != 2) return false;  // Only support bit_depth=2

    uint32_t row_bytes = (w + 7) / 8;
    uint32_t plane_size = row_bytes * h;

    const uint8_t *ptr = kp_data + 16;

    // Read LSB plane
    uint32_t lsb_size = read_u32(ptr); ptr += 4;
    const uint8_t *lsb_data;
    if (compression) {
        if (plane_size > KP_DECOMP_BUF_SIZE) return false;
        int dec = LZ4_decompress_safe((const char *)ptr, (char *)decomp_buf, lsb_size, plane_size);
        if (dec < 0) return false;
        lsb_data = decomp_buf;
        ptr += lsb_size;
    } else {
        lsb_data = ptr;
        ptr += lsb_size;
    }

    // Read MSB plane
    uint32_t msb_size = read_u32(ptr); ptr += 4;
    const uint8_t *msb_data;
    if (compression) {
        // Need second buffer — use stack for small assets, or decompress in-place
        // For now: use a second static buffer (UI assets are small)
        static uint8_t decomp_buf2[KP_DECOMP_BUF_SIZE];
        if (plane_size > KP_DECOMP_BUF_SIZE) return false;
        int dec = LZ4_decompress_safe((const char *)ptr, (char *)decomp_buf2, msb_size, plane_size);
        if (dec < 0) return false;
        msb_data = decomp_buf2;
    } else {
        msb_data = ptr;
    }

    // Blit: MSB=BW plane (level >> 1), LSB=GS plane (level & 1)
    // blit_gs expects: (fb_bw, fb_gs, x, y, bw_data, gs_data, w, h)
    if (fb_gs) {
        blit_gs(fb_bw, fb_gs, dst_x, dst_y, msb_data, lsb_data, w, h);
    } else {
        blit_bw(fb_bw, dst_x, dst_y, msb_data, w, h);
    }

    return true;
}

bool kp_blit_centered(const uint8_t *kp_data, uint32_t kp_size,
                       uint8_t *fb_bw, uint8_t *fb_gs,
                       int center_x, int bottom_y) {
    uint16_t w, h;
    if (!kp_get_dimensions(kp_data, &w, &h)) return false;
    int x = center_x - w / 2;
    int y = bottom_y - h;
    return kp_blit(kp_data, kp_size, fb_bw, fb_gs, x, y);
}

bool kp_get_dimensions(const uint8_t *kp_data, uint16_t *w, uint16_t *h) {
    if (memcmp(kp_data, KP_MAGIC, 4) != 0) return false;
    *w = read_u16(kp_data + 4);
    *h = read_u16(kp_data + 6);
    return true;
}
```

```c
#include <stdint.h>
#include <stdbool.h>

#define KB_MAX_TITLE   128
#define KB_MAX_AUTHOR   64
#define KB_MAX_PAGES   500
#define KB_MAX_ASSETS   20

// Asset types (from web's asset-gen.js)
#define KB_ASSET_FONT_NAME     0x00
#define KB_ASSET_BOOK_TITLE    0x01
#define KB_ASSET_BOOK_AUTHOR   0x02
#define KB_ASSET_COVER         0x03
#define KB_ASSET_CHAPTER_NAME  0x04
#define KB_ASSET_COVER_THUMB   0x05

typedef struct {
    char title[KB_MAX_TITLE];
    char author[KB_MAX_AUTHOR];
    uint16_t page_count;
    uint8_t chapter_count;
    uint8_t asset_count;
} kb_metadata_t;

typedef struct {
    uint32_t offset;   // file offset to KP blob
    uint32_t size;     // KP blob size
} kb_page_entry_t;

    uint8_t index;
    uint16_t height;
    uint32_t offset;
```

- [ ] **Step 2: Implement kb_reader.c**

Create `firmware/src/kb_reader.c`:
```c
#include "kb_reader.h"
#include <stdio.h>
#include <string.h>
#include "esp_log.h"

static const char *TAG = "kb_reader";

// KB v2 magic: 'K' 'B' 0x00 0x02
static const uint8_t KB_MAGIC[] = { 0x4B, 0x42, 0x00, 0x02 };

static FILE *_file = NULL;
static kb_metadata_t _meta;
static kb_page_entry_t _pages[KB_MAX_PAGES];
static kb_asset_entry_t _assets[KB_MAX_ASSETS];
static uint16_t _page_count = 0;
static uint8_t _asset_count = 0;

static inline uint16_t read_u16(const uint8_t *p) { return p[0] | (p[1] << 8); }
static inline uint32_t read_u32(const uint8_t *p) { return p[0] | (p[1] << 8) | (p[2] << 16) | (p[3] << 24); }

bool kb_open(const char *path) {
    kb_close();

    _file = fopen(path, "rb");
    if (!_file) {
        ESP_LOGE(TAG, "Cannot open: %s", path);
        return false;
    }

    // Read 32-byte header
    uint8_t hdr[32];
    if (fread(hdr, 1, 32, _file) != 32) goto fail;
    if (memcmp(hdr, KB_MAGIC, 4) != 0) {
        ESP_LOGE(TAG, "Invalid magic");
        goto fail;
    }

    _page_count = read_u16(hdr + 0x04);
    uint8_t chapter_count = hdr[0x06];
    _asset_count = read_u16(hdr + 0x18);

    uint32_t page_table_off = read_u32(hdr + 0x0C);
    uint32_t asset_off = read_u32(hdr + 0x14);
    uint32_t meta_off = read_u32(hdr + 0x1A);
    uint16_t meta_size = read_u16(hdr + 0x1E);

    if (_page_count > KB_MAX_PAGES) _page_count = KB_MAX_PAGES;
    if (_asset_count > KB_MAX_ASSETS) _asset_count = KB_MAX_ASSETS;

    // Read page table (8 bytes per entry: offset + size)
    fseek(_file, page_table_off, SEEK_SET);
    for (int i = 0; i < _page_count; i++) {
        uint8_t entry[8];
        if (fread(entry, 1, 8, _file) != 8) goto fail;
        _pages[i].offset = read_u32(entry);
        _pages[i].size = read_u32(entry + 4);
    }

    // Read asset index (12 bytes per entry)
    fseek(_file, asset_off, SEEK_SET);
    for (int i = 0; i < _asset_count; i++) {
        uint8_t entry[12];
        if (fread(entry, 1, 12, _file) != 12) goto fail;
        _assets[i].type = entry[0];
        _assets[i].index = entry[1];
        _assets[i].height = read_u16(entry + 2);
        _assets[i].offset = read_u32(entry + 4);
        _assets[i].size = read_u32(entry + 8);
    }

    // Read metadata JSON → extract title + author
    memset(&_meta, 0, sizeof(_meta));
    _meta.page_count = _page_count;
    _meta.chapter_count = chapter_count;
    _meta.asset_count = _asset_count;

    if (meta_size > 0 && meta_size < 2048) {
        char meta_buf[2048];
        fseek(_file, meta_off, SEEK_SET);
        int read = fread(meta_buf, 1, meta_size, _file);
        if (read > 0) {
            meta_buf[read] = '\0';
            // Simple JSON parse for "title" and "author"
            // (no JSON lib — just find "title":"..." pattern)
            char *t = strstr(meta_buf, "\"title\":\"");
            if (t) {
                t += 9;
                char *end = strchr(t, '"');
                if (end) {
                    int len = end - t;
                    if (len >= KB_MAX_TITLE) len = KB_MAX_TITLE - 1;
                    memcpy(_meta.title, t, len);
                    _meta.title[len] = '\0';
                }
            }
            char *a = strstr(meta_buf, "\"author\":\"");
            if (a) {
                a += 10;
                char *end = strchr(a, '"');
                if (end) {
                    int len = end - a;
                    if (len >= KB_MAX_AUTHOR) len = KB_MAX_AUTHOR - 1;
                    memcpy(_meta.author, a, len);
                    _meta.author[len] = '\0';
                }
            }
        }
    }

    ESP_LOGI(TAG, "Opened: %s (%d pages, %d assets)", path, _page_count, _asset_count);
    return true;

fail:
    fclose(_file);
    _file = NULL;
    return false;
}

void kb_close(void) {
    if (_file) {
        fclose(_file);
        _file = NULL;
    }
    _page_count = 0;
    _asset_count = 0;
}

bool kb_is_open(void) { return _file != NULL; }

const kb_metadata_t *kb_get_metadata(void) { return &_meta; }

uint32_t kb_load_asset(uint8_t type, uint8_t index, uint8_t *buf, uint32_t buf_size) {
    if (!_file) return 0;
    for (int i = 0; i < _asset_count; i++) {
        if (_assets[i].type == type && _assets[i].index == index) {
            if (_assets[i].size > buf_size) return 0;
            fseek(_file, _assets[i].offset, SEEK_SET);
            int read = fread(buf, 1, _assets[i].size, _file);
            return (read == _assets[i].size) ? _assets[i].size : 0;
        }
    }
    return 0;
}

uint32_t kb_load_page(uint32_t page_num, uint8_t *buf, uint32_t buf_size) {
    if (!_file || page_num >= _page_count) return 0;
    if (_pages[page_num].size > buf_size) return 0;
    fseek(_file, _pages[page_num].offset, SEEK_SET);
    int read = fread(buf, 1, _pages[page_num].size, _file);
    return (read == _pages[page_num].size) ? _pages[page_num].size : 0;
}
```

```c
#include "home.h"
#include "ui_assets.h"
#include "kp_decode.h"
#include "kb_reader.h"
#include "blit.h"
#include <x4/display.h>
#include <x4/input.h>
#include <x4/power.h>
#include <string.h>
#include <dirent.h>
#include "esp_log.h"

static const char *TAG = "home";

// Display dimensions (portrait)
#define PW 480
#define PH 800
#define SAFE_X 8
#define SAFE_W (PW - SAFE_X * 2)

// Layout constants (from emulator layout.js)
#define COVER_TOP_Y     64
#define COVER_W         300
#define COVER_H         400
#define TITLE_BOTTOM_Y  603
#define AUTHOR_Y        611
#define PROGRESS_LABEL_Y 696
#define PROGRESS_BAR_Y  700
#define INFO_X          28
#define HELP_BAR_Y      752
#define HELP_SLOT_W     116
#define HELP_SLOT_H     39

// Framebuffers
static uint8_t fb_bw[X4_DISPLAY_FB_SIZE];
static uint8_t fb_gs[X4_DISPLAY_FB_SIZE];

// Book list
#define MAX_BOOKS 32
typedef struct {
    char path[80];
    char title[128];
    char author[64];
    uint16_t page_count;
    uint8_t progress;  // 0-100
} book_info_t;

static book_info_t books[MAX_BOOKS];
static int book_count = 0;
static int current_book = -1;

// Cached assets from current book's .kb (loaded once per book switch)
#define ASSET_BUF_SIZE 32768  // 32KB for cover thumb + title + author KP blobs
static uint8_t asset_buf[ASSET_BUF_SIZE];
static uint32_t cover_thumb_offset, cover_thumb_size;
static uint32_t title_offset, title_size;
static uint32_t author_offset, author_size;

// Battery (mock for testing, later: read ADC)
static uint8_t battery_percent = 85;

// ── Book scanning ──

void home_scan_books(void) {
    book_count = 0;
    current_book = -1;

    DIR *dir = opendir("/sd");
    if (!dir) {
        ESP_LOGW(TAG, "Cannot open /sd");
        return;
    }

    struct dirent *ent;
    while ((ent = readdir(dir)) && book_count < MAX_BOOKS) {
        int len = strlen(ent->d_name);
        if (len < 4) continue;
        if (strcmp(ent->d_name + len - 3, ".kb") != 0) continue;

        char path[80];
        snprintf(path, sizeof(path), "/sd/%s", ent->d_name);

        if (kb_open(path)) {
            const kb_metadata_t *meta = kb_get_metadata();
            strncpy(books[book_count].path, path, sizeof(books[book_count].path) - 1);
            strncpy(books[book_count].title, meta->title, sizeof(books[book_count].title) - 1);
            strncpy(books[book_count].author, meta->author, sizeof(books[book_count].author) - 1);
            books[book_count].page_count = meta->page_count;
            books[book_count].progress = 0;  // TODO: read from NVS
            kb_close();
            book_count++;
        }
    }
    closedir(dir);

    if (book_count > 0) current_book = 0;
    ESP_LOGI(TAG, "Found %d books", book_count);
}

// ── Load current book's assets ──

static void load_book_assets(void) {
    if (current_book < 0 || current_book >= book_count) return;

    if (!kb_open(books[current_book].path)) return;

    // Load cover thumb, title, author into asset_buf sequentially
    uint32_t offset = 0;

    cover_thumb_size = kb_load_asset(KB_ASSET_COVER_THUMB, 0,
                                      asset_buf + offset, ASSET_BUF_SIZE - offset);
    cover_thumb_offset = offset;
    offset += cover_thumb_size;

    title_size = kb_load_asset(KB_ASSET_BOOK_TITLE, 0,
                                asset_buf + offset, ASSET_BUF_SIZE - offset);
    title_offset = offset;
    offset += title_size;

    author_size = kb_load_asset(KB_ASSET_BOOK_AUTHOR, 0,
                                 asset_buf + offset, ASSET_BUF_SIZE - offset);
    author_offset = offset;
    offset += author_size;

    kb_close();
    ESP_LOGI(TAG, "Loaded assets: cover=%lu title=%lu author=%lu",
             cover_thumb_size, title_size, author_size);
}

// ── Render helpers ──

static void blit_ui_asset(ui_asset_id_t id, int x, int y) {
    const ui_asset_t *a = &ui_lang[id];
    kp_blit(a->data, a->size, fb_bw, fb_gs, x, y);
}

static void blit_ui_asset_centered(ui_asset_id_t id, int cx, int bottom_y) {
    const ui_asset_t *a = &ui_lang[id];
    kp_blit_centered(a->data, a->size, fb_bw, fb_gs, cx, bottom_y);
}

static void draw_battery_fill(int x, int y, uint8_t percent) {
    // Battery frame from asset, fill computed
    const ui_asset_t *bat = &ui_lang[ASSET_BATTERY_FRAME];
    int tip_w = 3, sw = 2;
    int bw = bat->w - 2 - tip_w;  // body width minus pad
    int bh = bat->h - 2;          // body height minus pad
    int inner_w = bw - sw * 2;
    int inner_h = bh - sw * 2;
    int fill_w = (inner_w * percent + 50) / 100;
    if (fill_w > 0) {
        // Fill from right side (battery fills right-to-left)
        int fx = x + 1 + tip_w + sw + (inner_w - fill_w);
        int fy = y + 1 + sw;
        blit_fill_rect(fb_bw, fx, fy, fill_w, inner_h, 0);
        blit_fill_rect(fb_gs, fx, fy, fill_w, inner_h, 0);
    }
}

static void draw_progress_fill(int x, int y, uint8_t percent) {
    if (percent <= 0) return;
    const ui_asset_t *prog = &ui_lang[ASSET_PROGRESS_FRAME];
    int pad = 3;  // from definitions.yaml progress.frame.pad
    int inner_w = prog->w - 2 - pad * 2;
    int inner_h = prog->h - 2 - pad * 2;
    int fill_w = (inner_w * percent + 50) / 100;
    if (fill_w > 0) {
        int fx = x + 1 + pad;
        int fy = y + 1 + pad;
        blit_fill_rect(fb_bw, fx, fy, fill_w, inner_h, 0);
        blit_fill_rect(fb_gs, fx, fy, fill_w, inner_h, 0);
    }
}

// ── Screen callbacks ──

void home_enter(void) {
    if (book_count == 0) home_scan_books();
    if (current_book >= 0) load_book_assets();
}

void home_render(void) {
    // Clear to white
    memset(fb_bw, 0xFF, X4_DISPLAY_FB_SIZE);
    memset(fb_gs, 0xFF, X4_DISPLAY_FB_SIZE);

    // Battery
    blit_ui_asset(ASSET_BATTERY_FRAME, 433, 18);
    draw_battery_fill(433, 18, battery_percent);

    if (current_book >= 0 && current_book < book_count) {
        book_info_t *b = &books[current_book];

        // Cover thumbnail (centered)
        int thumb_x = (PW - COVER_W) / 2;  // 90
        if (cover_thumb_size > 0) {
            kp_blit(asset_buf + cover_thumb_offset, cover_thumb_size,
                    fb_bw, fb_gs, thumb_x, COVER_TOP_Y);
            // Border (2px)
            blit_fill_rect(fb_bw, thumb_x - 3, COVER_TOP_Y - 3, COVER_W + 6, 2, 0);
            blit_fill_rect(fb_bw, thumb_x - 3, COVER_TOP_Y + COVER_H + 1, COVER_W + 6, 2, 0);
            blit_fill_rect(fb_bw, thumb_x - 3, COVER_TOP_Y - 3, 2, COVER_H + 6, 0);
            blit_fill_rect(fb_bw, thumb_x + COVER_W + 1, COVER_TOP_Y - 3, 2, COVER_H + 6, 0);
            // Same on gs
            blit_fill_rect(fb_gs, thumb_x - 3, COVER_TOP_Y - 3, COVER_W + 6, 2, 0);
            blit_fill_rect(fb_gs, thumb_x - 3, COVER_TOP_Y + COVER_H + 1, COVER_W + 6, 2, 0);
            blit_fill_rect(fb_gs, thumb_x - 3, COVER_TOP_Y - 3, 2, COVER_H + 6, 0);
            blit_fill_rect(fb_gs, thumb_x + COVER_W + 1, COVER_TOP_Y - 3, 2, COVER_H + 6, 0);
        }

        // Arrows (centered in gap between safe edge and cover)
        int left_gap_center = SAFE_X + (thumb_x - SAFE_X) / 2;
        int right_gap_center = thumb_x + COVER_W + (PW - SAFE_X - thumb_x - COVER_W) / 2;
        int arrow_cy = COVER_TOP_Y + COVER_H / 2;
        const ui_asset_t *al = &ui_lang[ASSET_ARROW_LEFT];
        const ui_asset_t *ar = &ui_lang[ASSET_ARROW_RIGHT];
        blit_ui_asset(ASSET_ARROW_LEFT, left_gap_center - al->w / 2, arrow_cy - al->h / 2);
        blit_ui_asset(ASSET_ARROW_RIGHT, right_gap_center - ar->w / 2, arrow_cy - ar->h / 2);

        // Title (bottom-left anchored at TITLE_BOTTOM_Y)
        if (title_size > 0) {
            uint16_t tw, th;
            if (kp_get_dimensions(asset_buf + title_offset, &tw, &th)) {
                kp_blit(asset_buf + title_offset, title_size,
                        fb_bw, fb_gs, INFO_X, TITLE_BOTTOM_Y - th);
            }
        }

        // Author
        if (author_size > 0) {
            kp_blit(asset_buf + author_offset, author_size,
                    fb_bw, fb_gs, INFO_X, AUTHOR_Y);
        }

        // Progress label
        uint8_t prog = b->progress;
        ui_asset_id_t prog_id;
        if (prog == 0) prog_id = ASSET_PROGRESS_UNREAD;
        else if (prog >= 100) prog_id = ASSET_PROGRESS_FINISHED;
        else prog_id = ASSET_PROGRESS_1 + prog - 1;
        blit_ui_asset_centered(prog_id, PW / 2, PROGRESS_LABEL_Y);

        // Progress bar frame + fill
        blit_ui_asset(ASSET_PROGRESS_FRAME, INFO_X - 1, PROGRESS_BAR_Y);
        draw_progress_fill(INFO_X - 1, PROGRESS_BAR_Y, prog);
    } else {
        // No books
        blit_ui_asset_centered(ASSET_EMPTY_NO_BOOK, PW / 2, PH / 2 - 10);
        blit_ui_asset_centered(ASSET_EMPTY_NO_BOOK_SUB, PW / 2, PH / 2 + 20);
    }

    // Help bar divider
    blit_hline(fb_bw, SAFE_X, HELP_BAR_Y - 1, SAFE_W, 0);
    blit_hline(fb_gs, SAFE_X, HELP_BAR_Y - 1, SAFE_W, 0);

    // Help bar slots
    for (int i = 0; i < 4; i++) {
        ui_asset_id_t ids[] = { ASSET_HELP_READ, ASSET_HELP_LIBRARY,
                                ASSET_HELP_GALLERY, ASSET_HELP_SETTINGS };
        int sx = SAFE_X + i * HELP_SLOT_W;
        blit_ui_asset(ids[i], sx, HELP_BAR_Y);
    }

    // Send to display
    x4_display_render_grayscale(fb_bw, fb_gs, X4_REFRESH_FULL);
}

ui_state_t home_update(x4_input_event_t *evt) {
    if (evt->type != X4_EVT_PRESS && evt->type != X4_EVT_REPEAT) return STATE_HOME;

    switch (evt->button) {
        case X4_BTN_LEFT:
            if (book_count > 0) {
                current_book = (current_book - 1 + book_count) % book_count;
                load_book_assets();
                home_render();
            }
            break;
        case X4_BTN_RIGHT:
            if (book_count > 0) {
                current_book = (current_book + 1) % book_count;
                load_book_assets();
                home_render();
            }
            break;
        case X4_BTN_UP:
            // Mock battery change for testing
            battery_percent = (battery_percent + 10) % 110;
            if (battery_percent > 100) battery_percent = 0;
            home_render();
            break;
        case X4_BTN_DOWN:
            // Mock battery decrease
            if (battery_percent >= 10) battery_percent -= 10;
            else battery_percent = 100;
            home_render();
            break;
        default:
            break;
    }
    return STATE_HOME;
}

void home_exit(void) {
    // Nothing to clean up
}
```
