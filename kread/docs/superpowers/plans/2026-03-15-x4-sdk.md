# x4 SDK Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create a general-purpose hardware abstraction SDK for the Xteink X4 e-reader, porting the proven Papyrix display driver to pure C / ESP-IDF.

**Architecture:** PlatformIO library at `kread/firmware/lib/x4/` with 4 modules (display, input, sd, power) behind a single `x4_init()` entry point. Display driver is a line-by-line port of Papyrix `EInkDisplay.cpp`. All modules use ESP-IDF APIs directly — no Arduino.

**Tech Stack:** C (no C++), ESP-IDF 5.x, PlatformIO, ESP32-C3

**Spec:** `kread/docs/superpowers/specs/2026-03-15-x4-sdk-display-driver-design.md`

- [ ] **Step 3: Implement low-level SPI functions**

Port from Papyrix `sendCommand`, `sendData`, `waitWhileBusy`. Use `spi_device_transmit()` with manual CS.

```c
static void send_command(uint8_t cmd)
{
    gpio_set_level(PIN_DC, 0);
    gpio_set_level(PIN_CS, 0);
    spi_transaction_t t = { .length = 8, .tx_buffer = &cmd };
    spi_device_transmit(x4_spi_display_dev, &t);
    gpio_set_level(PIN_CS, 1);
}

static void send_data_byte(uint8_t val)
{
    gpio_set_level(PIN_DC, 1);
    gpio_set_level(PIN_CS, 0);
    spi_transaction_t t = { .length = 8, .tx_buffer = &val };
    spi_device_transmit(x4_spi_display_dev, &t);
    gpio_set_level(PIN_CS, 1);
}

static void send_data(const uint8_t *data, size_t len)
{
    gpio_set_level(PIN_DC, 1);
    gpio_set_level(PIN_CS, 0);
    while (len > 0) {
        size_t chunk = (len > 4096) ? 4096 : len;
        spi_transaction_t t = { .length = chunk * 8, .tx_buffer = data };
        spi_device_transmit(x4_spi_display_dev, &t);
        data += chunk;
        len -= chunk;
    }
    gpio_set_level(PIN_CS, 1);
}

static void wait_busy(const char *comment)
{
    uint32_t start = xTaskGetTickCount() * portTICK_PERIOD_MS;
    while (gpio_get_level(PIN_BUSY) == 1) {
        vTaskDelay(pdMS_TO_TICKS(1));
        uint32_t elapsed = xTaskGetTickCount() * portTICK_PERIOD_MS - start;
        if (elapsed > 10000) {
            ESP_LOGE(TAG, "BUSY timeout%s%s", comment ? ": " : "", comment ? comment : "");
            break;
        }
    }
    if (comment) {
        uint32_t elapsed = xTaskGetTickCount() * portTICK_PERIOD_MS - start;
        ESP_LOGD(TAG, "Wait done: %s (%lums)", comment, elapsed);
    }
}
```

- [ ] **Step 4: Implement `set_ram_area()` — CRITICAL, port from Papyrix exactly**

Port from Papyrix `setRamArea()` lines 285-318. Use 4-byte pixel coordinates for X range.

```c
static void set_ram_area(uint16_t x, uint16_t y, uint16_t w, uint16_t h)
{
    // Reverse Y (gates reversed on GDEQ0426T82)
    y = X4_DISPLAY_HEIGHT - y - h;

    // Data entry mode: X increment, Y decrement
    send_command(CMD_DATA_ENTRY_MODE);
    send_data_byte(0x01);

    // RAM X range — PIXEL coordinates, 4 bytes
    send_command(CMD_SET_RAM_X_RANGE);
    send_data_byte(x % 256);
    send_data_byte(x / 256);
    send_data_byte((x + w - 1) % 256);
    send_data_byte((x + w - 1) / 256);

    // RAM Y range — PIXEL coordinates, 4 bytes
    send_command(CMD_SET_RAM_Y_RANGE);
    send_data_byte((y + h - 1) % 256);
    send_data_byte((y + h - 1) / 256);
    send_data_byte(y % 256);
    send_data_byte(y / 256);

    // RAM X counter — 2 bytes
    send_command(CMD_SET_RAM_X_COUNTER);
    send_data_byte(x % 256);
    send_data_byte(x / 256);

    // RAM Y counter — 2 bytes
    send_command(CMD_SET_RAM_Y_COUNTER);
    send_data_byte((y + h - 1) % 256);
    send_data_byte((y + h - 1) / 256);
}
```

```c
static void init_display_controller(void)
{
    ESP_LOGI(TAG, "Initializing SSD1677...");

    // Soft reset
    send_command(CMD_SOFT_RESET);
    wait_busy("soft_reset");

    // Temperature sensor: internal
    send_command(CMD_TEMP_SENSOR);
    send_data_byte(0x80);

    // Booster soft start
    send_command(CMD_BOOSTER_SOFT_START);
    send_data_byte(0xAE);
    send_data_byte(0xC7);
    send_data_byte(0xC3);
    send_data_byte(0xC0);
    send_data_byte(0x40);

    // Driver output control: height=480, SM=1, TB=0
    send_command(CMD_DRIVER_OUTPUT);
    send_data_byte((X4_DISPLAY_HEIGHT - 1) % 256);
    send_data_byte((X4_DISPLAY_HEIGHT - 1) / 256);
    send_data_byte(0x02);

    // Border waveform
    send_command(CMD_BORDER_WAVEFORM);
    send_data_byte(0x01);

    // Set full screen RAM area
    set_ram_area(0, 0, X4_DISPLAY_WIDTH, X4_DISPLAY_HEIGHT);

    // Auto clear BW RAM
    send_command(CMD_AUTO_WRITE_BW_RAM);
    send_data_byte(0xF7);
    wait_busy("auto_clear_bw");

    // Auto clear RED RAM
    send_command(CMD_AUTO_WRITE_RED_RAM);
    send_data_byte(0xF7);
    wait_busy("auto_clear_red");
```

- [ ] **Step 7: Implement `x4_display_init()`**

```c
void x4_display_init(const x4_display_config_t *config)
{
    ESP_LOGI(TAG, "Display init (buffer_mode=%d)", config->buffer_mode);

    is_screen_on = false;
    custom_lut_active = false;
    in_grayscale_mode = false;
    buffer_mode = config->buffer_mode;

    // Set up framebuffers
    framebuffer = framebuffer0;
    memset(framebuffer0, 0xFF, X4_DISPLAY_FB_SIZE);

    if (buffer_mode == X4_DISPLAY_DUAL_BUFFER) {
        framebuffer1 = (uint8_t *)malloc(X4_DISPLAY_FB_SIZE);
        ESP_ERROR_CHECK(framebuffer1 ? ESP_OK : ESP_ERR_NO_MEM);
        memset(framebuffer1, 0xFF, X4_DISPLAY_FB_SIZE);
        framebuffer_active = framebuffer1;
        ESP_LOGI(TAG, "Dual buffer: 2 x %d bytes", X4_DISPLAY_FB_SIZE);
    } else {
        ESP_LOGI(TAG, "Single buffer: %d bytes", X4_DISPLAY_FB_SIZE);
    }

    // GPIO for DC, RST, BUSY (CS configured by x4.c SPI init)
    gpio_config_t out_conf = {
        .pin_bit_mask = (1ULL << PIN_DC) | (1ULL << PIN_RST),
        .mode = GPIO_MODE_OUTPUT,
    };
    gpio_config(&out_conf);

    gpio_config_t in_conf = {
        .pin_bit_mask = (1ULL << PIN_BUSY),
        .mode = GPIO_MODE_INPUT,
    };
    gpio_config(&in_conf);

    gpio_set_level(PIN_CS, 1);
    gpio_set_level(PIN_DC, 1);

    // Hardware reset
    gpio_set_level(PIN_RST, 1);
    vTaskDelay(pdMS_TO_TICKS(20));
    gpio_set_level(PIN_RST, 0);
    vTaskDelay(pdMS_TO_TICKS(2));
    gpio_set_level(PIN_RST, 1);
    vTaskDelay(pdMS_TO_TICKS(20));

    // Initialize controller
    init_display_controller();

    ESP_LOGI(TAG, "Display initialized (%dx%d)", X4_DISPLAY_WIDTH, X4_DISPLAY_HEIGHT);
}
```

- [ ] **Step 2: Implement `refresh_display()`**

Port from Papyrix `refreshDisplay()` lines 544-598. This is the most critical function — display mode bits must match exactly.

```c
static void refresh_display(x4_display_refresh_t mode, bool turn_off)
{
    // CTRL1: buffer comparison mode
    send_command(CMD_UPDATE_CTRL1);
    send_data_byte((mode == X4_REFRESH_FAST) ? CTRL1_NORMAL : CTRL1_BYPASS_RED);

    uint8_t display_mode = 0x00;

    // Power up if screen is off
    if (!is_screen_on) {
        is_screen_on = true;
        display_mode |= 0xC0;  // CLOCK_ON + ANALOG_ON
    }

    // Power down if requested
    if (turn_off) {
        is_screen_on = false;
        display_mode |= 0x03;  // ANALOG_OFF + CLOCK_OFF
    }

    // Refresh mode specific bits
    if (mode == X4_REFRESH_FULL) {
        display_mode |= 0x34;
    } else if (mode == X4_REFRESH_HALF) {
        // Write high temp for faster refresh
        send_command(CMD_WRITE_TEMP);
        send_data_byte(0x5A);
        display_mode |= 0xD4;
    } else {  // FAST
        display_mode |= custom_lut_active ? 0x0C : 0x1C;
    }

    // Execute refresh
    send_command(CMD_UPDATE_CTRL2);
    send_data_byte(display_mode);
    send_command(CMD_MASTER_ACTIVATION);

    const char *mode_str = (mode == X4_REFRESH_FULL) ? "full"
                         : (mode == X4_REFRESH_HALF) ? "half" : "fast";
    ESP_LOGD(TAG, "Refresh 0x%02X (%s)", display_mode, mode_str);
    wait_busy(mode_str);
}
```

```c
void x4_display_update(x4_display_refresh_t mode, bool turn_off)
{
    // Auto-upgrade FAST to HALF if screen is off
    if (!is_screen_on && mode == X4_REFRESH_FAST) {
        mode = X4_REFRESH_HALF;
    }

    // Revert grayscale if active
    x4_display_grayscale_revert();

    // Set full screen RAM area
    set_ram_area(0, 0, X4_DISPLAY_WIDTH, X4_DISPLAY_HEIGHT);

    if (mode != X4_REFRESH_FAST) {
        // Full/Half: write to both BW and RED
        write_ram_buffer(CMD_WRITE_RAM_BW, framebuffer, X4_DISPLAY_FB_SIZE);
        write_ram_buffer(CMD_WRITE_RAM_RED, framebuffer, X4_DISPLAY_FB_SIZE);
    } else {
        // Fast: write BW only (new frame)
        write_ram_buffer(CMD_WRITE_RAM_BW, framebuffer, X4_DISPLAY_FB_SIZE);
        if (buffer_mode == X4_DISPLAY_DUAL_BUFFER) {
            // RED = previous frame (always correct in dual buffer)
            write_ram_buffer(CMD_WRITE_RAM_RED, framebuffer_active, X4_DISPLAY_FB_SIZE);
        }
        // Single buffer: RED RAM already contains previous frame from post-sync
    }

    // Swap buffers (dual buffer mode)
    if (buffer_mode == X4_DISPLAY_DUAL_BUFFER) {
        uint8_t *temp = framebuffer;
        framebuffer = framebuffer_active;
        framebuffer_active = temp;
    }

    // Refresh
    refresh_display(mode, turn_off);

    // Single buffer: sync RED RAM after refresh
    if (buffer_mode == X4_DISPLAY_SINGLE_BUFFER) {
        set_ram_area(0, 0, X4_DISPLAY_WIDTH, X4_DISPLAY_HEIGHT);
        write_ram_buffer(CMD_WRITE_RAM_RED, framebuffer, X4_DISPLAY_FB_SIZE);
    }
}
```

```c
void x4_display_update_window(uint16_t x, uint16_t y, uint16_t w, uint16_t h, bool turn_off)
{
    if (x + w > X4_DISPLAY_WIDTH || y + h > X4_DISPLAY_HEIGHT) {
        ESP_LOGE(TAG, "Window bounds exceed display");
        return;
    }
    if (x % 8 != 0 || w % 8 != 0) {
        ESP_LOGE(TAG, "Window x/w must be byte-aligned (multiple of 8)");
        return;
    }

    // Auto-upgrade: windowed update needs valid RED RAM for differential comparison.
    // If screen is off, RED RAM may be stale — do a full-screen HALF refresh first.
    if (!is_screen_on) {
        ESP_LOGW(TAG, "Screen off — upgrading windowed update to full HALF refresh");
        x4_display_update(X4_REFRESH_HALF, false);
        // Now screen is on, RED RAM is current, proceed with window
    }

    x4_display_grayscale_revert();

    uint16_t win_w_bytes = w / 8;
    uint32_t win_size = win_w_bytes * h;

    uint8_t *win_buf = (uint8_t *)malloc(win_size);
    if (!win_buf) {
        ESP_LOGE(TAG, "Window buffer alloc failed (%lu bytes)", win_size);
        return;
    }

    // Extract window from framebuffer
    for (uint16_t row = 0; row < h; row++) {
        uint32_t src = (y + row) * X4_DISPLAY_WIDTH_BYTES + (x / 8);
        uint32_t dst = row * win_w_bytes;
        memcpy(&win_buf[dst], &framebuffer[src], win_w_bytes);
    }

    // Write to BW RAM
    set_ram_area(x, y, w, h);
    write_ram_buffer(CMD_WRITE_RAM_BW, win_buf, win_size);

    // Dual buffer: extract previous frame window for RED RAM
    if (buffer_mode == X4_DISPLAY_DUAL_BUFFER) {
        uint8_t *prev_buf = (uint8_t *)malloc(win_size);
        if (prev_buf) {
            for (uint16_t row = 0; row < h; row++) {
                uint32_t src = (y + row) * X4_DISPLAY_WIDTH_BYTES + (x / 8);
                uint32_t dst = row * win_w_bytes;
                memcpy(&prev_buf[dst], &framebuffer_active[src], win_w_bytes);
            }
            write_ram_buffer(CMD_WRITE_RAM_RED, prev_buf, win_size);
            free(prev_buf);
        }
    }

    // Fast refresh the window
    refresh_display(X4_REFRESH_FAST, turn_off);

    // Single buffer: sync RED RAM for this window
    if (buffer_mode == X4_DISPLAY_SINGLE_BUFFER) {
        set_ram_area(x, y, w, h);
        write_ram_buffer(CMD_WRITE_RAM_RED, win_buf, win_size);
    }

    free(win_buf);
}
```

```c
#include <string.h>
#include "esp_adc/adc_oneshot.h"
#include "driver/gpio.h"
#include "esp_log.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

// Pins
#define ADC_CH1         ADC_CHANNEL_1  // GPIO1: BACK, CONFIRM, LEFT, RIGHT
#define ADC_CH2         ADC_CHANNEL_2  // GPIO2: UP, DOWN
#define POWER_GPIO      3

// ADC thresholds (from Papyrix, proven on device)
#define ADC_NO_BUTTON   3800
static const int ADC1_THRESHOLDS[] = { 3800, 3100, 2090, 750 };
static const x4_button_t ADC1_BUTTONS[] = { X4_BTN_BACK, X4_BTN_CONFIRM, X4_BTN_LEFT, X4_BTN_RIGHT };
static const int ADC2_THRESHOLDS[] = { 3800, 1120 };
static const x4_button_t ADC2_BUTTONS[] = { X4_BTN_UP, X4_BTN_DOWN };

// Debounce
#define DEBOUNCE_MS     5

// Directional buttons get repeat; non-directional get long-press
static inline bool is_directional(x4_button_t btn) {
    return btn == X4_BTN_LEFT || btn == X4_BTN_RIGHT
        || btn == X4_BTN_UP   || btn == X4_BTN_DOWN;
}

// Event ring buffer
#define EVT_QUEUE_SIZE  16
static x4_input_event_t evt_queue[EVT_QUEUE_SIZE];
static int evt_head = 0;
static int evt_tail = 0;
static int evt_count = 0;

// Config
static x4_input_config_t cfg;

// Per-button state
static bool btn_down[X4_BTN_COUNT];
static uint32_t btn_down_time[X4_BTN_COUNT];
static uint32_t btn_debounce_time[X4_BTN_COUNT];
static bool btn_last_raw[X4_BTN_COUNT];
static bool btn_long_fired[X4_BTN_COUNT];
static uint32_t btn_last_repeat[X4_BTN_COUNT];

static adc_oneshot_unit_handle_t adc_handle;

// Exported for power.c to share ADC unit (ESP-IDF only allows one handle per unit)
adc_oneshot_unit_handle_t x4_adc_handle = NULL;

static void push_event(x4_button_t btn, x4_event_type_t type)
{
    if (evt_count >= EVT_QUEUE_SIZE) {
        // Drop oldest
        evt_tail = (evt_tail + 1) % EVT_QUEUE_SIZE;
        evt_count--;
    }
    evt_queue[evt_head].button = btn;
    evt_queue[evt_head].type = type;
    evt_head = (evt_head + 1) % EVT_QUEUE_SIZE;
    evt_count++;
}

void x4_input_init(const x4_input_config_t *config)
{
    cfg = config ? *config : (x4_input_config_t)X4_INPUT_CONFIG_DEFAULT;

    memset(btn_down, 0, sizeof(btn_down));
    memset(btn_last_raw, 0, sizeof(btn_last_raw));
    memset(btn_long_fired, 0, sizeof(btn_long_fired));
    evt_head = evt_tail = evt_count = 0;

    // ADC init
    adc_oneshot_unit_init_cfg_t adc_cfg = { .unit_id = ADC_UNIT_1 };
    ESP_ERROR_CHECK(adc_oneshot_new_unit(&adc_cfg, &adc_handle));

    adc_oneshot_chan_cfg_t chan_cfg = {
        .atten = ADC_ATTEN_DB_12,
        .bitwidth = ADC_BITWIDTH_12,
    };
    ESP_ERROR_CHECK(adc_oneshot_config_channel(adc_handle, ADC_CH1, &chan_cfg));
    ESP_ERROR_CHECK(adc_oneshot_config_channel(adc_handle, ADC_CH2, &chan_cfg));
    // Also configure battery channel (GPIO0 = ADC_CHANNEL_0) for power.c
    ESP_ERROR_CHECK(adc_oneshot_config_channel(adc_handle, ADC_CHANNEL_0, &chan_cfg));
    x4_adc_handle = adc_handle;

    // Power button GPIO
    gpio_config_t pwr_cfg = {
        .pin_bit_mask = (1ULL << POWER_GPIO),
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_ENABLE,
    };
    gpio_config(&pwr_cfg);

    ESP_LOGI(TAG, "Input initialized (7 buttons)");
}

static x4_button_t decode_adc(int mv, const int *thresholds, const x4_button_t *buttons, int count)
{
    if (mv >= ADC_NO_BUTTON) return X4_BTN_NONE;
    for (int i = 0; i < count - 1; i++) {
        if (mv >= thresholds[i + 1]) return buttons[i];
    }
    return buttons[count - 1];
}

void x4_input_poll(void)
{
    uint32_t now = xTaskGetTickCount() * portTICK_PERIOD_MS;

    // Read raw button state
    bool raw[X4_BTN_COUNT];
    memset(raw, 0, sizeof(raw));

    // ADC1
    int adc1_raw = 0;
    adc_oneshot_read(adc_handle, ADC_CH1, &adc1_raw);
    int mv1 = adc1_raw * 3300 / 4095;
    x4_button_t b1 = decode_adc(mv1, ADC1_THRESHOLDS, ADC1_BUTTONS, 4);
    if (b1 != X4_BTN_NONE) raw[b1] = true;

    // ADC2
    int adc2_raw = 0;
    adc_oneshot_read(adc_handle, ADC_CH2, &adc2_raw);
    int mv2 = adc2_raw * 3300 / 4095;
    x4_button_t b2 = decode_adc(mv2, ADC2_THRESHOLDS, ADC2_BUTTONS, 2);
    if (b2 != X4_BTN_NONE) raw[b2] = true;

    // Power button (active low)
    if (gpio_get_level(POWER_GPIO) == 0) raw[X4_BTN_POWER] = true;

    // Process each button
    for (int i = 0; i < X4_BTN_COUNT; i++) {
        // Debounce: only accept change after stable for DEBOUNCE_MS
        if (raw[i] != btn_last_raw[i]) {
            btn_debounce_time[i] = now;
            btn_last_raw[i] = raw[i];
        }

        bool stable = (now - btn_debounce_time[i]) >= DEBOUNCE_MS;
        if (!stable) continue;

        bool was_down = btn_down[i];
        bool is_down = raw[i];

        if (is_down && !was_down) {
            // Press
            btn_down[i] = true;
            btn_down_time[i] = now;
            btn_long_fired[i] = false;
            btn_last_repeat[i] = now;
            push_event((x4_button_t)i, X4_EVT_PRESS);
        } else if (!is_down && was_down) {
            // Release
            btn_down[i] = false;
            push_event((x4_button_t)i, X4_EVT_RELEASE);
        } else if (is_down) {
            // Held — check long press / repeat
            uint32_t held = now - btn_down_time[i];

            if (is_directional((x4_button_t)i)) {
                // Directional: repeat
                if (held >= cfg.repeat_start_ms) {
                    if (now - btn_last_repeat[i] >= cfg.repeat_interval_ms) {
                        btn_last_repeat[i] = now;
                        push_event((x4_button_t)i, X4_EVT_REPEAT);
                    }
                }
            } else {
                // Non-directional: long press (once)
                if (!btn_long_fired[i] && held >= cfg.long_press_ms) {
                    btn_long_fired[i] = true;
                    push_event((x4_button_t)i, X4_EVT_LONG_PRESS);
                }
            }
        }
    }
}

bool x4_input_next_event(x4_input_event_t *event)
{
    if (evt_count == 0) return false;
    *event = evt_queue[evt_tail];
    evt_tail = (evt_tail + 1) % EVT_QUEUE_SIZE;
    evt_count--;
    return true;
}
```

```c
#include <x4/power.h>
#include "esp_log.h"
#include "esp_sleep.h"
#include "esp_adc/adc_oneshot.h"
#include "esp_adc/adc_cali.h"
#include "esp_adc/adc_cali_scheme.h"
#include "driver/gpio.h"

static const char *TAG = "x4.power";

#define USB_GPIO        20  // UART0_RXD
#define POWER_GPIO      3

// Shared ADC handle from input.c (input owns ADC_UNIT_1, configures battery channel too)
extern adc_oneshot_unit_handle_t x4_adc_handle;

static adc_cali_handle_t adc_cali = NULL;

void x4_power_init(void)
{
    // USB detection GPIO
    gpio_config_t usb_cfg = {
        .pin_bit_mask = (1ULL << USB_GPIO),
        .mode = GPIO_MODE_INPUT,
    };
    gpio_config(&usb_cfg);

    // ADC calibration for accurate battery voltage reading
    // The polynomial was derived from calibrated mV readings (Papyrix uses analogReadMilliVolts)
    adc_cali_curve_fitting_config_t cali_cfg = {
        .unit_id = ADC_UNIT_1,
        .atten = ADC_ATTEN_DB_12,
        .bitwidth = ADC_BITWIDTH_12,
    };
    esp_err_t err = adc_cali_create_scheme_curve_fitting(&cali_cfg, &adc_cali);
    if (err != ESP_OK) {
        ESP_LOGW(TAG, "ADC calibration not available: %s", esp_err_to_name(err));
    }

    ESP_LOGI(TAG, "Power initialized");
}

int x4_power_battery_percent(void)
{
    if (!x4_adc_handle) return -1;

    int raw = 0;
    adc_oneshot_read(x4_adc_handle, ADC_CHANNEL_0, &raw);

    // Get calibrated millivolts (matches Papyrix analogReadMilliVolts behavior)
    int mv;
    if (adc_cali) {
        adc_cali_raw_to_voltage(adc_cali, raw, &mv);
    } else {
        mv = raw * 3300 / 4095;  // fallback, less accurate
    }

    // Voltage divider (2x)
    double volts = (mv * 2.0) / 1000.0;

    // Polynomial from CrossPoint/Papyrix BatteryMonitor
    double pct = -144.9390 * volts * volts * volts
               + 1655.8629 * volts * volts
               - 6158.8520 * volts
               + 7501.3202;

    if (pct < 0) pct = 0;
    if (pct > 100) pct = 100;
    return (int)(pct + 0.5);
}

bool x4_power_usb_connected(void)
{
    return gpio_get_level(USB_GPIO) == 1;
}

x4_wakeup_reason_t x4_power_wakeup_reason(void)
{
    esp_sleep_wakeup_cause_t cause = esp_sleep_get_wakeup_cause();
    switch (cause) {
        case ESP_SLEEP_WAKEUP_GPIO:
        case ESP_SLEEP_WAKEUP_EXT0:
            return X4_WAKEUP_POWER_BUTTON;
        default:
            if (cause == ESP_SLEEP_WAKEUP_UNDEFINED) return X4_WAKEUP_RESET;
            return X4_WAKEUP_OTHER;
    }
}

void x4_power_deep_sleep(void)
{
    ESP_LOGI(TAG, "Entering deep sleep...");
    // Wake on power button (GPIO3, active low = wake on low)
    esp_deep_sleep_enable_gpio_wakeup(1ULL << POWER_GPIO, ESP_GPIO_WAKEUP_GPIO_LOW);
    esp_deep_sleep_start();
}
```

ADC handle sharing: `input.c` owns `ADC_UNIT_1`, configures all channels including battery (ADC_CHANNEL_0), and exports `x4_adc_handle`. `power.c` uses `extern x4_adc_handle` — no separate ADC init needed. ADC calibration (`adc_cali_curve_fitting`) is used for accurate mV conversion matching Papyrix's `analogReadMilliVolts()`.

```c
#include "serial.h"

// Portrait dimensions (user-facing, mapped from native 800x480)
#define PORTRAIT_W  480
#define PORTRAIT_H  800

static void portrait_set_pixel(int x, int y, bool black)
{
    if (x < 0 || x >= PORTRAIT_W || y < 0 || y >= PORTRAIT_H) return;
    // Portrait (x,y) → native landscape (nx,ny)
    int nx = y;
    int ny = PORTRAIT_W - 1 - x;
    int byte_idx = ny * (X4_DISPLAY_WIDTH / 8) + (nx / 8);
    int bit_idx = 7 - (nx % 8);
    uint8_t *fb = x4_display_framebuffer();
    if (black) {
        fb[byte_idx] &= ~(1 << bit_idx);
    } else {
        fb[byte_idx] |= (1 << bit_idx);
    }
}

static void portrait_fill_rect(int x, int y, int w, int h, bool black)
{
    for (int py = y; py < y + h && py < PORTRAIT_H; py++) {
        for (int px = x; px < x + w && px < PORTRAIT_W; px++) {
            portrait_set_pixel(px, py, black);
        }
    }
}

static void draw_char(int x, int y, char c, bool black)
{
    if (c < 32 || c > 126) c = '?';
    const uint8_t *glyph = font8x16[c - 32];
    for (int row = 0; row < 16; row++) {
        uint8_t bits = glyph[row];
        for (int col = 0; col < 8; col++) {
            if (bits & (0x80 >> col)) {
                portrait_set_pixel(x + col, y + row, black);
            }
        }
    }
}

static void draw_string(int x, int y, const char *str, bool black)
{
    while (*str) { draw_char(x, y, *str, black); x += 8; str++; }
}

static void draw_centered(int y, const char *str, bool black)
{
    int x = (PORTRAIT_W - strlen(str) * 8) / 2;
    draw_string(x, y, str, black);
}

static void draw_test_screen(const char *pressed)
{
    x4_display_clear(0xFF);

    draw_centered(20, "kread button test (x4 SDK)", true);
    portrait_fill_rect(40, 50, PORTRAIT_W - 80, 1, true);

    draw_string(30, 700, "Bottom: BACK CONFIRM LEFT RIGHT", true);
    draw_string(30, 730, "Side: UP DOWN POWER", true);

    if (pressed && pressed[0]) {
        draw_centered(380, pressed, true);
    } else {
        draw_centered(380, "Press any button", true);
    }
}

void app_main(void)
{
    ESP_LOGI(TAG, "kread firmware v%s starting...", KREAD_VERSION);

    serial_init();

    x4_config_t cfg = {
        .display = { .buffer_mode = X4_DISPLAY_DUAL_BUFFER },
        .input   = X4_INPUT_CONFIG_DEFAULT,
        .sd      = X4_SD_CONFIG_DEFAULT,
    };
    x4_init(&cfg);

    draw_test_screen(NULL);
    x4_display_update(X4_REFRESH_FULL, true);

    ESP_LOGI(TAG, "Ready. Press buttons to test.");

    static const char *btn_names[] = {
        "BACK", "CONFIRM", "LEFT", "RIGHT", "UP", "DOWN", "POWER"
    };

    while (1) {
        serial_poll();
        x4_input_poll();

        x4_input_event_t evt;
        while (x4_input_next_event(&evt)) {
            if (evt.type == X4_EVT_PRESS) {
                ESP_LOGI(TAG, "Button: %s", btn_names[evt.button]);
                draw_test_screen(btn_names[evt.button]);
                // turn_off=false: keep screen on so next FAST refresh is actually fast (~600ms)
                // If turn_off=true, screen powers off and next FAST auto-upgrades to HALF (~1720ms)
                x4_display_update(X4_REFRESH_FAST, false);
            }
        }

        vTaskDelay(pdMS_TO_TICKS(10));
    }
}
```
